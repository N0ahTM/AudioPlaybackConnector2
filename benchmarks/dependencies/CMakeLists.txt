cmake_minimum_required(VERSION 4.2)
project(ApcDependencyProbes LANGUAGES CXX)
if(NOT MSVC OR MSVC_VERSION LESS 1951)
    message(FATAL_ERROR "Dependency probes require MSVC 14.51 or newer")
endif()
string(REPLACE "/W3" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
set(CMAKE_DISABLE_PRECOMPILE_HEADERS ON)
find_package(CLI11 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
foreach(probe IN ITEMS baseline cli11 json spdlog)
    add_executable(probe_${probe} ${probe}.cpp)
    target_compile_options(probe_${probe} PRIVATE /std:c++latest /W4 /WX /permissive- /utf-8)
endforeach()
target_link_libraries(probe_cli11 PRIVATE CLI11::CLI11)
target_link_libraries(probe_json PRIVATE nlohmann_json::nlohmann_json)
target_link_libraries(probe_spdlog PRIVATE spdlog::spdlog)
